   powershell -NoProfile -ExecutionPolicy Bypass -Command `
     "Invoke-WebRequest -Uri 'http://server/api/download/install.ps1' -OutFile 'install.ps1' -UseBasicParsing; `
      .\install.ps1 -Server 'server:8080' -EnrollToken 'TOKEN'"
